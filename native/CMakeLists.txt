cmake_minimum_required(VERSION 3.18)
project(LogixTagRelayNative LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This first adapter bridge targets Windows x64.")
endif()
set(OpENer_PLATFORM WIN32 CACHE STRING "" FORCE)
set(OpENer_Device_Config_Vendor_Id 0 CACHE STRING "ODVA Vendor ID; 0 for the development adapter")
set(OpENer_Device_Config_Device_Name "LogixTagRelay Adapter" CACHE STRING "" FORCE)
set(OPENER_ETHERNET_BUFFER_SIZE 1024 CACHE STRING "" FORCE)
add_subdirectory(OpENer/source opener-build EXCLUDE_FROM_ALL)
# Replace only the build target's application callbacks; the pinned dependency stays unmodified.
set_property(TARGET SAMPLE_APP PROPERTY SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/relay_application.c")
get_target_property(OPENER_INCLUDES SAMPLE_APP INCLUDE_DIRECTORIES)
add_library(LogixTagRelayOpener SHARED relay_host.c)
target_include_directories(LogixTagRelayOpener PRIVATE ${OPENER_INCLUDES} "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(SAMPLE_APP PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(LogixTagRelayOpener PRIVATE RESTRICT=__restrict WIN32 PC_OPENER_ETHERNET_BUFFER_SIZE=1024)
target_link_libraries(LogixTagRelayOpener PRIVATE PLATFORM_GENERIC WIN32PLATFORM CIP Utils SAMPLE_APP ENET_ENCAP NVDATA ws2_32 Iphlpapi)
